function finalizarPagamento() { // ===================================== // PEGAR DADOS DO CLIENTE // ===================================== const nomeElemento = document.getElementById("nome"); const telefoneElemento = document.getElementById("telefone"); const enderecoElemento = document.getElementById("endereco"); const numeroElemento = document.getElementById("numero"); const cidadeElemento = document.getElementById("cidade"); // Verificar se os campos existem if ( !nomeElemento || !telefoneElemento || !enderecoElemento || !numeroElemento || !cidadeElemento ) { alert( "❌ Erro: não foi possível encontrar os campos do cliente." ); return; } // Pegar valores const nome = nomeElemento.value.trim(); const telefone = telefoneElemento.value.trim(); const endereco = enderecoElemento.value.trim(); const numero = numeroElemento.value.trim(); const cidade = cidadeElemento.value.trim(); // ===================================== // VERIFICAR DADOS // ===================================== if ( !nome || !telefone || !endereco || !numero || !cidade ) { alert( "⚠️ Preencha todos os dados do cliente." ); return; } // ===================================== // PEGAR FORMA DE PAGAMENTO // ===================================== const pagamento = document.querySelector( 'input[name="pagamento"]:checked' ); if (!pagamento) { alert( "💳 Escolha uma forma de pagamento." ); return; } // ===================================== // RECUPERAR CARRINHO // ===================================== const dadosCarrinho = localStorage.getItem( "carrinhoCompra" ); let carrinho = []; try { carrinho = dadosCarrinho ? JSON.parse(dadosCarrinho) : []; } catch (erro) { console.error( "Erro ao ler o carrinho:", erro ); alert( "❌ O carrinho possui dados inválidos. Volte para a página de compra e tente novamente." ); return; } // ===================================== // VERIFICAR CARRINHO // ===================================== if ( !Array.isArray(carrinho) || carrinho.length === 0 ) { alert( "🛒 Seu carrinho está vazio." ); return; } // ===================================== // CALCULAR TOTAL // ===================================== let total = Number(pedido.total) || 0; if (total === 0 && Array.isArray(pedido.produtos)) { total = 0; pedido.produtos.forEach(produto => { const preco = Number(produto.preco) || 0; const quantidade = Number(produto.quantidade) || 0; total += preco * quantidade; }); } }; // ===================================== // CRIAR PEDIDO // ===================================== const pedido = { nome: nome, telefone: telefone, endereco: endereco, numero: numero, cidade: cidade, pagamento: pagamento.value, produtos: carrinho, total: total, data: new Date().toISOString() }; // ===================================== // SALVAR PEDIDO // ===================================== localStorage.setItem( "pedidoTemporario", JSON.stringify(pedido) ); // ===================================== // IR PARA FINALIZAÇÃO // ===================================== window.location.href = "finalizacao.html"; }
